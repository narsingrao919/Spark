# Write the sql query to get the movies with the most number of ratings in comedy genre in sql
​
highest_movie_ratingDf = spark.sql(
"""
select m.title, count(*) as rating_count from movies m join rating r on m.movieId = r.movieId
where m.genres like '%Comedy%' 
group by m.title
order by rating_count desc limit 1
"""
)
